\begin{figure}[H]
	\centering
	\begin{tikzpicture}[cross/.style={draw, cross out, minimum size=2*(#1-1pt), inner sep=0pt, outer sep=0pt}, x=10mm, y=10mm,
		>=latex,   %Voreinstellung für Pfeilspitzen
		font=\footnotesize, scale=0.8]
		%Raster zeichnen
		\draw [color=gray!50]  [step=10mm] (-5.3,-3.3) grid (5.3,5.3);
		% Achsen zeichnen
		\draw[->,thick] (-5.5,0) -- (5.8,0) node[left, below] {\normalsize $x$};
		\draw[->,thick] (0,-4.5) -- (0,5.8) node[above, left=4pt] {\normalsize $y$};
		% Achsen beschriften
		\foreach \x in {-5,...,-2,-1,1,2,...,5} \draw (\x,-.1) -- (\x,.1) node[right=3pt,below=7pt] {$\x$};
		\foreach \y in {-3,...,-2,-1,1,2,...,5} \draw (-.1,\y) -- (.1,\y) node[left=7pt] {$\y$};
		% Besserer Ursprung
		\draw[color=black] (0pt,-7pt) node[left] {$0$};
		%Funktionen zeichnen:
		\clip(-6.3,-3.3) rectangle (5.3,5.3); %Bildbereich
		\draw[red,thick,domain=-6.3:5.3,samples=200] plot (\x, {-3/4*(\x)+1.5});
		\draw[forestgreen,thick,domain=-6.3:5.3,samples=200] plot (\x, {2/3*(\x)+1});
		\draw[plotblue,thick,domain=-6.3:5.3,samples=200] plot (\x, {-2});
		%Beschriftung
		\node[red]  at (-3.2,4.5) {\normalsize$f$};
		\node[forestgreen]  at (-1.2,0.8) {\normalsize$g$};
		\node[plotblue]  at (3.2,-1.8) {\normalsize$h$};
		%Punkte
		\draw[fill=red] (-2,3) circle (2.5pt);
		\draw[fill=red] (2,0) circle (2.5pt);
		\draw[fill=forestgreen] (3,3) circle (2.5pt);
		\draw[fill=forestgreen] (-3,-1) circle (2.5pt);
		\draw[fill=plotblue] (-4,-2) circle (2.5pt);
		\draw[fill=plotblue] (1,-2) circle (2.5pt);
	\end{tikzpicture}
\end{figure}
