\end{parts}

\question[\totalpoints]  %Barth SE K5.3 A2
\addpoints
\begin{align*}
	5x^2-8x&=0 && \vert\div x\\
	5x-8&=0 &&\vert+8\:\vert\div 5\\
	x&=\frac{8}{5}
\end{align*}
\begin{parts}
	\part[1] Kennzeichne alle Fehler im obigen Lösungsweg.
	\part[2] Löse die Gleichung korrekt.
\end{parts}
