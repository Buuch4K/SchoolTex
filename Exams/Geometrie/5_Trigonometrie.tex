	\begin{tabular}{l|c|c}
		& richtig & falsch\\\hline
		$\sin{\alpha}=\sin{180^\circ-\alpha}$ & & \\
		$\cos{180^\circ-\beta}=\cos{180^\circ+\beta}$ & & \\
		$\tan{\alpha}>0$ für $90^\circ<\alpha<180^\circ$ & & \\
		$\cos{\alpha}=\sin{\alpha}$ für $\alpha=\frac{\pi}{6}$ & & \\
		$\sin{360^\circ-\beta}=\sin{180^\circ-\beta}$ & & \\

	\end{tabular}
\end{table}
