\question[\totalpoints]
\addpoints
\begin{parts}
    \part[2] Beschreibe in eigenen Worten, was ein Logarithmus ist und inwiefern er mit der Exponentialfunktion verbunden ist.
    \part[1] Erkläre einem Laien in zwei Sätzen, den Unterschied zwischen einer Polynomialgleichung und einer Exponentialgleichung. Veranschauliche deine Erklärung mit einem passenden Beispiel.
\end{parts}
\ifthenelse{\boolean{printanswers}}
{\begin{solution}
		(a) $\log{a}{b}$ ist der Exponent $x$, mit dem man $a$ potenzieren muss, um $b$ zu erhalten: $\log{a}{b}=x\same a^x=b$. Die Logarithmusfunktion $y=\log{a}{x}$ ist die Umkehrfunktion der Exponentialfunktion $y=a^x$.\\
		(b) Bei einer Polynomialgleichung steht die Unbekannte in der Basis (z.\,B. $x^3=8$, Lösung durch Wurzelziehen), bei einer Exponentialgleichung im Exponenten (z.\,B. $3^x=8$, Lösung mit dem Logarithmus).
\end{solution}}
{\antwort{6}}

    \part Welcher der folgenden Ausdrücke passt nicht dazu? Begründe deine Antwort.
    \[\log{4}{2}\quad\log{10}{100}\quad\log{2}{8}\quad\log{3}{81}\]
\end{parts}
\ifthenelse{\boolean{printanswers}}

		(c) $\log{4}{2}=\frac{1}{2}$, $\log{10}{0.01}=-2$, $\log{2}{\frac{1}{8}}=-3$, $\log{3}{\frac{1}{81}}=-4$: $\log{4}{2}$ passt nicht, da er als einziger positiv (und keine ganze Zahl) ist.\\

\question[4]
Begründe, welche der folgenden Ausdrücke wahr sind und korrigiere die falschen Aussagen.
\begin{parcols}[2]
    \part $\log{a}{u+v}=\log{a}{u}+\log{a}{v}$
    \part $\log{a}{u-v}=\frac{\log{a}{u}}{\log{a}{v}}$
    \part $\log{2}{x}+\log{3}{y}=\log{2}{xy}$
    \part $\log{a}{b}=\frac{\ln{a}}{\ln{b}}$
\end{parcols}
\ifthenelse{\boolean{printanswers}}
{\begin{solution}
		Alle vier Aussagen sind falsch. Korrekt ist:\\
		(a) $\log{a}{u\*v}=\log{a}{u}+\log{a}{v}$ \quad (b) $\log{a}{\frac{u}{v}}=\log{a}{u}-\log{a}{v}$\\
		(c) $\log{2}{x}+\log{2}{y}=\log{2}{xy}$ (gleiche Basen nötig) \quad (d) $\log{a}{b}=\frac{\ln{b}}{\ln{a}}$
\end{solution}}
{\antwort{6}}

\question[4]
Claudia will diese Gleichung lösen: \[5^x=0.2^{3x}\*\exp{x-1}.\] Sie wendet den natürlichen Logarithmus auf beiden Seiten an und erhält \[x\*\ln{5}=3x\*\ln{0.2}\*(x-1).\] Hat sie alles richtig gemacht?
\begin{itemize}
	\item Falls JA, löse die Gleichung fertig.
	\item Falls NEIN, erkläre den Fehler und löse die Gleichung vollständig korrekt.
\end{itemize}
\ifthenelse{\boolean{printanswers}}
{\begin{solution}
		Nein. Der Logarithmus eines Produkts ist die Summe der Logarithmen, nicht ihr Produkt. Richtig ist $x\*\ln{5}=3x\*\ln{0.2}+(x-1)$.\\
		Mit $\ln{0.2}=-\ln{5}$: $x\*\ln{5}+3x\*\ln{5}-x=-1\ \Rightarrow\ x\*\left(4\*\ln{5}-1\right)=-1\ \Rightarrow\ x=\frac{-1}{4\*\ln{5}-1}\approx-0.184$
\end{solution}}
{\antwort{8}}
