\end{parts}
\ifthenelse{\boolean{printanswers}}
{\begin{solution}
		$\left[x^2 + x\right]_1^b = b^2 + b - 2 = 10 \Rightarrow b^2 + b - 12 = 0 \Rightarrow (b-3)(b+4) = 0$.\\
		Wegen $b > 1$ folgt $b = 3$.
\end{solution}}
