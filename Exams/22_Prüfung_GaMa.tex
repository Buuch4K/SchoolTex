\question[\totalpoints]

Bestimme die erste Ableitung der folgenden Funktionen schrittweise von Hand. Gib in deinem Lösungsweg an, welche der folgenden Ableitungsregeln (PR = Produktregel, QR = Quotientenregel, KR = Kettenregel) verwendet wird. Die Resultate müssen \textbf{nicht} vereinfacht werden.
\begin{parts}
	\part[2] $\displaystyle f(x)=p\*\sin{x}-\cos{p\*x}$
	\part[2] $\displaystyle f(x)=\frac{4x-2+\ln{x}}{2x^2+\exp{x}}$
	\part[2] $\displaystyle f(x)=\exp{\frac{x^2+3x-1}{2}}$
	\part[2] $\displaystyle f(x)=\sqrt{\frac{2(x^6-x^3)}{x^2}}$
\end{parts}

\begin{parts}
	\part[1] $\displaystyle\frac{\d{\ }}{\d{x}}\left(z\*f(x)\right)=z\*\left(\frac{\d{\ }}{\d{x}}f(x)\right)+f(x)\*\left(\frac{\d{\ }}{\d{x}}z\right)$
	\part[1] $\displaystyle\frac{\d{\ }}{\d{x}}\left(z\*f(x)\right)=\left(\frac{\d{\ }}{\d{x}}z\right)\*\left(\frac{\d{\ }}{\d{x}}f(x)\right)$
	\part[1] $\displaystyle\frac{\d{\ }}{\d{x}}\left(z\*f(x)\right)=z\*\left(\frac{\d{\ }}{\d{x}}f(x)\right)$
\end{parts}
